BOOL FASTCALL Scheduler::Load(Fileio *fio, int ver)
{
	size_t sz;
	Event *first;

	ASSERT(this);
	ASSERT(fio);
	ASSERT(ver >= 0x200);
	ASSERT_DIAG();

	LOG0(Log::Normal, "????????????");

	// ????????????????????????????????????????
	first = sch.first;

	// ???????????????????????????????????????????????????????????????????
	if (!fio->Read(&sz, sizeof(sz))) {
		return FALSE;
	}
	if (sz != sizeof(breakp)) {
		return FALSE;
	}

	// ??????????????????????????????????????????????????
	if (!fio->Read(breakp, (int)sz)) {
		return FALSE;
	}

	// ???????????????????????????????????????????????????????????????????????
	if (!fio->Read(&sz, sizeof(sz))) {
		return FALSE;
	}
	if (sz != sizeof(scheduler_t)) {
		return FALSE;
	}

	// ?????????????????????????????????????????????????
	if (!fio->Read(&sch, (int)sz)) {
		return FALSE;
	}

	// ??????????????????????????????????????????????
	if (!fio->Read(CycleTable, sizeof(CycleTable))) {
		return FALSE;
	}

	// ???????????????????????????????????????????
	sch.first = first;

	// ??????????????????????2.01??????????????????ma_active??????????????
	if (ver >= 0x0201) {
		if (!fio->Read(&dma_active, sizeof(dma_active))) {
			return FALSE;
		}
	}

	return TRUE;
}

//	??????????

DWORD FASTCALL Scheduler::Exec(DWORD hus)
{
	int cycle;
	DWORD result;
	DWORD dcycle;

	ASSERT(this);
	ASSERT(hus > 0);
	ASSERT_DIAG();

	// ????????????????????????????????????????????????
	if (!sch.check) {
		// ?????????????????????????????????????
#if defined(SCHEDULER_ASM)
		sch.one = GetMinEvent(hus);
#else
		sch.one = GetMinRemain(hus);
#endif	// SCHEDULER_ASM

		// sch.one + sch.time????????????????????????????????????????????????????????????????????????????
		ASSERT((sch.one + sch.time) < 0x1000);
		cycle = CycleTable[sch.one + sch.time];
		if (cycle > sch.cycle) {

			// ???????????????????????????????????????????????????????????????????????????
			cycle -= sch.cycle;
			if (!dma_active) {
				// ??????????
				result = cpu->Exec(cycle);
			}
			else {
				// DMAC??????????????????????????????????????
				dcycle = dmac->AutoDMA(cycle);
				if (dcycle != 0) {
					// ?????????????????????????????????????????????????
					result = cpu->Exec(dcycle);
				}
				else {
					// ???????????DMA???????????????
					result = cycle;
				}
			}

			// ?????????????????????????
			if (result < 0x80000000) {
				// sch.time, sch.cycle???????????
				sch.cycle += result;
				sch.time += sch.one;

				// ???????????????????
				ExecEvent(sch.one);

				if (sch.time < 200) {
					return sch.one;
				}

				// ?????Sync
				while (sch.time >= 200) {
					if ((DWORD)sch.cycle < sch.speed) {
						break;
					}
					sch.time -= 200;
					sch.cycle -= sch.speed;
				}

				// ????????????????????????????
				if (!sch.brk) {
					return sch.one;
				}

#if defined(SCHEDULER_LOG)
				LOG0(Log::Normal, "??????????????");
#endif	// SCHEDULER_LOG
				sch.brk = FALSE;
				return (DWORD)(sch.one | 0x80000000);
			}
			else {
				// ??????????????????
				result &= 0x7fffffff;

				if ((int)result > cycle) {
					// sch.time????h.cycle???????????
					sch.time += sch.one;
					sch.cycle += result;

					// ?????????????????????????
					ExecEvent(sch.one);

					while (sch.time >= 200) {
						if ((DWORD)sch.cycle < sch.speed) {
							break;
						}
						sch.time -= 200;
						sch.cycle -= sch.speed;
					}
					// ???????????????????????????????????????????
					return 0x80000000;
				}
				// ??????????????????????????????cpu????????????????????????????
				sch.cycle += result;
				// ???????????????????????????????????????????
				return 0x80000000;
			}
		}
		else {

			// ?????????????????????????????????????????????????????????????????
			sch.time += sch.one;
			ExecEvent(sch.one);

			if (sch.time < 200) {
				return sch.one;
			}

			// sch.time???????????
			while (sch.time >= 200) {
				if ((DWORD)sch.cycle < sch.speed) {
					break;
				}
				sch.time -= 200;
				sch.cycle -= sch.speed;
			}

			// ?????????????????????????????????????????????????
			return sch.one;
		}

	}

	// ????????????
	for (;;) {
		result = Trace(hus);

		switch (result) {
			// ?????????????????????????????????????????????????
			case 0:
				return sch.one;

			// ????????????????????????????????????
			case 1:
				if (sch.brk) {
#if defined(SCHEDULER_LOG)
					LOG0(Log::Normal, "??????????????");
#endif	// SCHEDULER_LOG
					sch.brk = FALSE;
					return 0x80000000;
				}
				if (IsBreak(cpu->GetPC()) != -1) {
					OnBreak(cpu->GetPC());
					return 0x80000000;
				}
				return sch.one;

			// ???????????????????????????????????????
			case 2:
				if (sch.brk) {
#if defined(SCHEDULER_LOG)
					LOG0(Log::Normal, "??????????????");
#endif	// SCHEDULER_LOG
					sch.brk = FALSE;
					return 0x80000000;
				}
				if (IsBreak(cpu->GetPC()) != -1) {
					OnBreak(cpu->GetPC());
					return 0x80000000;
				}
				break;

			// ??????????????????
			case 3:
				if (sch.brk) {
#if defined(SCHEDULER_LOG)
					LOG0(Log::Normal, "??????????????");
#endif	// SCHEDULER_LOG
					sch.brk = FALSE;
				}
				return 0x80000000;

			// ???????????????
			default:
				ASSERT(FALSE);
				return sch.one;
		}
	}
}

//---------------------------------------------------------------------------
//
//	??????????????
//
//---------------------------------------------------------------------------
DWORD FASTCALL Scheduler::Trace(DWORD hus)
{
	int cycle;
	DWORD result;

	ASSERT(this);
	ASSERT(hus > 0);
	ASSERT_DIAG();

	// ?????????????????????????????????????
#if defined(SCHEDULER_ASM)
	sch.one = GetMinEvent(hus);
#else
	sch.one = GetMinRemain(hus);
#endif	// SCHEDULER_ASM

	// sch.one + sch.time????????????????????????????????????????????????????????????????????????????
	ASSERT((sch.one + sch.time) < 0x1000);
	cycle = CycleTable[sch.one + sch.time];
	if (cycle <= sch.cycle) {
		// ??????????????????????????????????????????????????????????
		sch.time += sch.one;
		ExecEvent(sch.one);

		// sch.time???????????
		while (sch.time >= 200) {
			sch.time -= 200;
			sch.cycle -= sch.speed;
		}
		// ?????????????????????????????????????????????????
		return 0;
	}

	// ????????????????????????????????????????????????????????
	cycle -= sch.cycle;

	// 1??????????????????????????????????????????????????????????????????
	if (!dma_active) {
		// ??????????
		result = cpu->Exec(1);
	}
	else {
		// DMAC??????????????????????????????????????
		result = dmac->AutoDMA(1);
		if (result != 0) {
			result = cpu->Exec(result);
		}
		else {
			result = 1;
		}
	}
	if (result >= 0x80000000) {
		// ??????????????????
		return 3;
	}

	// result >= cycle??????????????????????????????????????????
	if ((int)result >= cycle) {
		// sch.time, sch.cycle???????????
		sch.cycle += result;
		sch.time += sch.one;

		// ???????????????????
		ExecEvent(sch.one);

		while (sch.time >= 200) {
			sch.time -= 200;
			sch.cycle -= sch.speed;
		}
		// ????????????????????????????????????
		return 1;
	}

	// ???????????????????????????????????????????????????????????????????????????????????
	// sch.cycle???????????
	sch.cycle += result;

	// ???????????????????????????????????????
	return 2;
}

void FASTCALL Scheduler::SetCPUSpeed(DWORD speed)
{
	int i;
	DWORD cycle;

	ASSERT(this);
	ASSERT(speed > 0);
	ASSERT_DIAG();

	LOG2(Log::Detail, "CPU???????????????????? %d.%02dMHz", speed / 100, (speed % 100));

	// CPU????????????????????????
	sch.speed = speed;

	// 0?????2048us???????????????0.5us?????????????????????????????????????????????????????????????????
	for (i=0; i<0x1000; i++) {
		cycle = (DWORD)i;
		cycle *= speed;
		cycle /= 200;
		CycleTable[i] = cycle;
	}
}

//---------------------------------------------------------------------------
//
//	???????????????????

//---------------------------------------------------------------------------
DWORD FASTCALL Scheduler::GetPassedTime() const
{
	DWORD hus;

	ASSERT(this);
	ASSERT_DIAG();

	// ????????????????????????????????????????0
	if (sch.exec) {
		return 0;
	}

	// ?????????????????????????????????pu_cylcle?????????????????????????
	hus = cpu->GetCycle() + sch.cycle;
	hus *= 200;
	hus /= sch.speed;
	hus -= sch.time;

	// one????????????????????????????????????????????
	if (sch.one < hus) {
		hus = sch.one;
	}

	// hus????????????????
	return hus;
}

void FASTCALL Scheduler::SetBreak(DWORD addr, BOOL enable)
{
	int i;
	BOOL flag;

	ASSERT(this);
	ASSERT(addr <= 0xffffff);
	ASSERT_DIAG();

#if defined(SCHEDULER_LOG)
	LOG2(Log::Normal, "????????????????????????????????????? $%06X enable=%d", addr, enable);
#endif	// SCHEDULER_LOG

	flag = FALSE;

	// ???????????????????????
	for (i=0; i<BreakMax; i++) {
		if (breakp[i].use) {
			if (breakp[i].addr == addr) {
				// ?????????????????????????
				breakp[i].enable = enable;
				flag = TRUE;
				break;
			}
		}
	}

	if (!flag) {
		// ???????????????????
		for (i=0; i<BreakMax; i++) {
			if (!breakp[i].use) {
				// ?????????????
				breakp[i].use = TRUE;
				breakp[i].addr = addr;
				breakp[i].enable = enable;
				breakp[i].time = 0;
				breakp[i].count = 0;
				break;
			}
		}
	}

	// ?????????????????????????????
	flag = FALSE;
	for (i=0; i<BreakMax; i++) {
		if (breakp[i].use) {
			if (breakp[i].enable) {
				// ???????????????????????????????????????????
				flag = TRUE;
				break;
			}
		}
	}
	sch.check = flag;
}

int FASTCALL Scheduler::IsBreak(DWORD addr, BOOL any) const
{
	int i;

	ASSERT(this);
	ASSERT(addr <= 0xffffff);
	ASSERT_DIAG();

	// ?????????????????????????????
	if (!sch.check) {
		return -1;
	}

	// ???????????????????????
	for (i=0; i<BreakMax; i++) {
		if (breakp[i].use) {
			if (breakp[i].addr == addr) {
				// ???????????????????????????????????????????????????
				if (any || breakp[i].enable) {
					return i;
				}
			}
		}
	}

	// ?????????????????????????????????????????????????????????????????
	return -1;
}

	}
	ASSERT(i < BreakMax);

	// ????????????????????????????????????????????
	breakp[i].time = GetTotalTime();
	breakp[i].count++;
}

//---------------------------------------------------------------------------
void FASTCALL Scheduler::AddEvent(Event *event)
{
	Event *p;

	ASSERT(this);
	ASSERT(event);
	ASSERT_DIAG();

#if defined(SCHEDULER_LOG)
	LOG4(Log::Normal, "??????????????????????? Device=%c%c%c%c",
					(char)(event->GetDevice()->GetID() >> 24),
					(char)(event->GetDevice()->GetID() >> 16),
					(char)(event->GetDevice()->GetID() >> 8),
					(char)(event->GetDevice()->GetID()));
	LOG1(Log::Normal, "??????????????????????? %s", event->GetDesc());
#endif	// SCHEDULER_LOG

	// ????????????????????????????
	if (!sch.first) {
		// ???????????????????????????
		sch.first = event;
		event->SetNextEvent(NULL);

#if defined(SCHEDULER_ASM)
		// ??????
		NotifyEvent(sch.first);
#endif	// SCHEDULER_ASM
		return;
	}

	// ?????????????????????????????????????
	p = sch.first;
	while (p->GetNextEvent()) {
		p = p->GetNextEvent();
	}

	// p????????????????????????????????????????????????????????????
	p->SetNextEvent(event);
	event->SetNextEvent(NULL);

#if defined(SCHEDULER_ASM)
	// ??????
	NotifyEvent(sch.first);
#endif	// SCHEDULER_ASM
}

//---------------------------------------------------------------------------
//
//	?????????????????????

//---------------------------------------------------------------------------
void FASTCALL Scheduler::DelEvent(Event *event)
{
	Event *p;
	Event *prev;

	ASSERT(this);
	ASSERT(event);
	ASSERT_DIAG();

#if defined(SCHEDULER_LOG)
	LOG4(Log::Normal, "????????????????????? Device=%c%c%c%c",
					(char)(event->GetDevice()->GetID() >> 24),
					(char)(event->GetDevice()->GetID() >> 16),
					(char)(event->GetDevice()->GetID() >> 8),
					(char)(event->GetDevice()->GetID()));
	LOG1(Log::Normal, "????????????????????? %s", event->GetDesc());
#endif	// SCHEDULER_LOG

	// ????????????????????????????
	if (sch.first == event) {
		// ???????????????????????????????ext??????????????????????????????????????????????????
		sch.first = event->GetNextEvent();
		event->SetNextEvent(NULL);

#if defined(SCHEDULER_ASM)
		// ??????
		NotifyEvent(sch.first);
#endif	// SCHEDULER_ASM
		return;
	}

	// ????????????????????????????????????????????????????????????
	p = sch.first;
	prev = p;
	while (p) {
		// ???????????????????????
		if (p == event) {
			prev->SetNextEvent(event->GetNextEvent());
			event->SetNextEvent(NULL);

#if defined(SCHEDULER_ASM)
			// ??????
			NotifyEvent(sch.first);
#endif	// SCHEDULER_ASM
			return;
		}

		// ??????????
		prev = p;
		p = p->GetNextEvent();
	}

}

		if (remain == 0) {

void FASTCALL Scheduler::ExecEvent(DWORD hus)
{
#if !defined(SCHEDULER_ASM)
	Event *p;
#endif	// !SCHEDULER_ASM

	ASSERT(this);
	ASSERT(hus >= 0);
	ASSERT_DIAG();

	// ??????????????????????????????
	sch.exec = TRUE;

	// ??????????????????????????????????????????????????????????
	sch.total += hus;
	sch.sound += hus;

#if defined(SCHEDULER_ASM)
	SubExecEvent(hus);
	sch.exec = FALSE;
#else

	// ????????????????????????????????????????
	p = sch.first;

	// ????????????????????????????????????????
	while (p) {
		p->Exec(hus);
		p = p->GetNextEvent();
	}

	// ??????????????????????????????????
	sch.exec = FALSE;
#endif
}
